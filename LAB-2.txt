---For teacher_new table---
**command for hive**

CREATE EXTERNAL TABLE IF NOT EXISTS teacher_tmp (
    Teacher_id INT,
    Teacher_name STRING,
    Email STRING,
    Dept_id STRING
)
COMMENT 'teacher_tmp'
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
LOCATION '/TeacherDetail/Teacher_Tmp';


CREATE EXTERNAL TABLE IF NOT EXISTS teacher_new (
    Teacher_id INT,
    Teacher_name STRING,
    Email STRING
)
PARTITIONED BY (Dept_id STRING)
STORED AS PARQUET
LOCATION '/TeacherDetail/Teacher_new';

**command for window command prompt**

docker cp "D:\4th year even sem\DDBMS lab\lab-2\Teacher_New.csv" myHadoop:/test/

**command for hive**
set hive.exec.dynamic.partition=true;
set hive.exec.dynamic.partition.mode=nonstrict;

INSERT OVERWRITE TABLE teacher_new PARTITION(Dept_id)SELECT Teacher_id,Teacher_name,Email,Dept_id from teacher_tmp;

SELECT * FROM teacher_new
WHERE Dept_id='1' or Dept_id='6';

show partitions teacher_new;
desc teacher-new;

**command for root**
hdfs dfs -ls /TeacherDetail/Teacher_new
hdfs dfs -ls /TeacherDetail/Teacher_new/dept_id=9
