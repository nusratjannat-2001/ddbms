create database Teacherinfo;
use teacherinfo;

CREATE EXTERNAL TABLE IF NOT EXISTS Teacher (
    Teacher_id INT,
    Teacher_name STRING,
    Email STRING
)
COMMENT 'Teacher_Details'
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/TeacherDetail/teacher';


CREATE EXTERNAL TABLE IF NOT EXISTS Course (
    course_id INT,
    course_name STRING,
    credit INT,
    dept_id INT,
    teacher_id INT
)

COMMENT 'Course information'
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/TeacherDetail/course';

CREATE EXTERNAL TABLE IF NOT EXISTS Department (
    dep_id INT,
    dept_name STRING,
    building STRING
)

COMMENT 'Dept Details'
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/TeacherDetail/department';
