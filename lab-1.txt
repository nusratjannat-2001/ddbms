CREATE DATABASE IF NOT EXISTS TeacherInfo;
show databases;
use teacherInfo;

CREATE EXTERNAL TABLE IF NOT EXISTS Department(
    Dept_Id INT,
    Dept_Name STRING,
    Building STRING
)
COMMENT 'Dept_Details'
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/TeacherDetail/Department';

desc Department;

CREATE EXTERNAL TABLE IF NOT EXISTS Teacher (
    Teacher_Id INT,
    Teacher_Name STRING,
    Email STRING,
    Dept_Id INT
)
COMMENT 'Teacher_Details'
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/TeacherDetail/Teacher';

desc Teacher;

CREATE EXTERNAL TABLE IF NOT EXISTS Course (
    course_Id INT,
    course_Name STRING,
    credit STRING,
    Dept_Id INT,
    teacher_id INT
)
COMMENT 'Teacher_Details'
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/TeacherDetail/Course';

desc Course;

show tables;
